! Lab 01 - OSPF Routing
! Router: R2
! LAN: 192.168.20.0/24
! Links: R1 and R3


en
conf t
hos R2
no ip domain-lookup

int gig 0/1
des link to R1
ip add 10.0.12.2 255.255.255.252
no shut
ex

int gig 0/2
des link to R3
ip add 10.0.23.1 255.255.255.252
no shut
ex

int gig 0/0
des LAN-2
ip add 192.168.20.1 255.255.255.0
no shut
ex

router ospf 1
rout-id 2.2.2.2
network 192.168.20.0 0.0.0.255 area 0
network 10.0.12.0 0.0.0.3 area 0
network 10.0.23.0 0.0.0.3 area 0
end
write mem

! Lab 01 - OSPF Routing
! Router: R1
! LAN: 192.168.10.0/24
! Transit link to R2: 10.0.12.0/30


ena
conf t
ho R1
no ip domain-lookup

int gig 0/0
des LAN-1
ip add 192.168.10.1 255.255.255.0
no shut
ex

int gig 0/1
des Link to R1
ip add 10.0.2.1 255.255.255.252
no shut
ex

router ospf 1
rout-id 1.1.1.1
network 192.168.10.0 0.0.0.255 area 0
netw 10.0.12.0 0.0.0.3 area 0
end
writ mem

! Lab 01 - OSPF Routing
! Router: R1
! LAN: 192.168.10.0/24
! Transit link to R2: 10.0.12.0/30


enable
configure terminal

hostname R1
no ip domain-lookup

interface GigabitEthernet0/0
 description LAN-1
 ip address 192.168.10.1 255.255.255.0
 no shutdown
exit

interface GigabitEthernet0/1
 description LINK-TO-R2
 ip address 10.0.12.1 255.255.255.252
 no shutdown
exit

router ospf 1
 router-id 1.1.1.1
 network 192.168.10.0 0.0.0.255 area 0
 network 10.0.12.0 0.0.0.3 area 0
exit

end
write memory
